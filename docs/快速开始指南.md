# 微前端子应用快速开始指南

## 🚀 5分钟快速集成子应用

本指南将帮助你在 5 分钟内快速创建并集成一个子应用到现有的微前端基座中。

## 📋 前置条件

- ✅ 基座应用已正常运行 (http://localhost:3000)
- ✅ Node.js >= 18.0.0
- ✅ npm >= 8.0.0

## 🎯 步骤概览

```
1. 创建子应用 (2分钟)
2. 配置 qiankun (1分钟)
3. 注册到基座 (1分钟)
4. 测试运行 (1分钟)
```

### 步骤 2: 配置 qiankun (1分钟)

#### Vue 应用配置

<details>
<summary><strong>Vue 配置文件</strong></summary>

**vite.config.ts**
```typescript
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import qiankun from 'vite-plugin-qiankun'

export default defineConfig({
  plugins: [
    vue(),
    qiankun('my-vue-app', { useDevMode: true })
  ],
  server: {
    port: 3001,  // 确保端口不冲突
    cors: true,
    origin: 'http://localhost:3001'
  }
})
```

**src/main.ts**
```typescript
import { createApp } from 'vue'
import { createRouter, createWebHistory } from 'vue-router'
import { renderWithQiankun, qiankunWindow } from 'vite-plugin-qiankun/dist/helper'
import App from './App.vue'

let app: any
let router: any

const routes = [
  { path: '/', component: () => import('./components/HelloWorld.vue') }
]

function render(props: any = {}) {
  const { container, routerBase } = props
  
  router = createRouter({
    history: createWebHistory(routerBase || '/my-vue-app/'),
    routes
  })
  
  app = createApp(App)
  app.use(router)
  
  const containerEl = container ? container.querySelector('#app') : '#app'
  app.mount(containerEl)
}

renderWithQiankun({
  mount(props) {
    render(props)
  },
  bootstrap() {},
  unmount() {
    app?.unmount()
  }
})

if (!qiankunWindow.__POWERED_BY_QIANKUN__) {
  render()
}
```

</details>

#### React 应用配置

<details>
<summary><strong>React 配置文件</strong></summary>

**vite.config.ts**
```typescript
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import qiankun from 'vite-plugin-qiankun'

export default defineConfig({
  plugins: [
    react(),
    qiankun('my-react-app', { useDevMode: true })
  ],
  server: {
    port: 3002,  // 确保端口不冲突
    cors: true,
    origin: 'http://localhost:3002'
  }
})
```

**src/main.tsx**
```typescript
import React from 'react'
import ReactDOM from 'react-dom/client'
import { BrowserRouter, Routes, Route } from 'react-router-dom'
import { renderWithQiankun, qiankunWindow } from 'vite-plugin-qiankun/dist/helper'

let root: ReactDOM.Root | null = null

const App = () => (
  <div>
    <h1>My React App</h1>
    <Routes>
      <Route path="/" element={<div>首页</div>} />
      <Route path="/about" element={<div>关于</div>} />
    </Routes>
  </div>
)

function render(props: any = {}) {
  const { container, routerBase } = props
  const containerEl = container ? container.querySelector('#root') : document.getElementById('root')
  
  if (!root) {
    root = ReactDOM.createRoot(containerEl!)
  }
  
  root.render(
    <BrowserRouter basename={routerBase || '/my-react-app'}>
      <App />
    </BrowserRouter>
  )
}

renderWithQiankun({
  mount(props) {
    render(props)
  },
  bootstrap() {},
  unmount() {
    root?.unmount()
    root = null
  }
})

if (!qiankunWindow.__POWERED_BY_QIANKUN__) {
  render()
}
```

</details>

### 步骤 3: 注册到基座 (1分钟)

编辑基座应用的微应用配置：

```typescript
// ad-engine/src/config/microApps.ts
export const devMicroApps: MicroApp[] = [
  // 现有的应用...
  
  // 添加你的新应用
  {
    name: 'my-vue-app',        // 或 'my-react-app'
    entry: 'http://localhost:3001',  // 或 3002
    container: '#subapp-viewport',
    activeRule: '/my-vue-app',       // 或 '/my-react-app'
    props: {
      routerBase: '/my-vue-app',     // 或 '/my-react-app'
    },
  },
];
```

在基座应用路由中添加：

```typescript
// ad-engine/src/router/index.tsx
export const router = createBrowserRouter([
  {
    path: '/',
    element: <Layout />,
    children: [
      // 现有路由...
      
      // 添加新的子应用路由
      {
        path: 'my-vue-app/*',     // 或 'my-react-app/*'
        element: <MicroAppContainer />,
      },
    ],
  },
]);
```

在导航中添加链接（可选）：

```typescript
// ad-engine/src/components/Layout.tsx
<nav className="header-nav">
  {/* 现有导航... */}
  
  <Link 
    to="/my-vue-app"  // 或 "/my-react-app"
    className={location.pathname.startsWith('/my-vue-app') ? 'active' : ''}
  >
    我的应用
  </Link>
</nav>
```

### 步骤 4: 测试运行 (1分钟)

```bash
# 终端 1: 启动基座应用 (如果未启动)
cd ad-engine
npm run dev

# 终端 2: 启动子应用
cd sub-apps/my-vue-app  # 或 my-react-app
npm run dev

# 访问测试
# 基座中访问: http://localhost:3000/my-vue-app
# 独立访问: http://localhost:3001 (或 3002)
```

```bash

lsof -i :3001

# 检查 CORS 配置

```

### 问题 2: 路由 404

```typescript
// 检查路由配置是否正确
// Vue: createWebHistory('/my-vue-app/')
// React: basename="/my-react-app"
```

### 问题 3: 样式丢失

```typescript
// 基座应用 qiankun 配置
start({
  sandbox: {
    strictStyleIsolation: false,
    experimentalStyleIsolation: true,
  }
});
```
